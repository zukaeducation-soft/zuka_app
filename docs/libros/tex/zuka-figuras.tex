% zuka-figuras.tex — ilustraciones de los libros. Uso en los capítulos: \figura{nombre}{pie de figura}.
% Cada dibujo es \zfig@<nombre>. Unidades en mm. Estilo: fichas con relieve, sombra suave, letra Fira.
\usetikzlibrary{shadows,shadings,decorations.pathmorphing}
\newcounter{zfig}\@addtoreset{zfig}{mundo}
\newcommand{\figura}[2]{\par\vspace{7pt}\begin{center}\stepcounter{zfig}\csname zfig@#1\endcsname\par\vspace{3pt}
  {\sffamily\footnotesize\color{zmute}\textbf{\color{zolive}Figura \themundo.\arabic{zfig}}\quad #2}\end{center}\vspace{2pt}}
\tikzset{
  zf/.style={x=1mm,y=1mm,font=\sffamily\scriptsize,text=zink},
  sombra/.style={drop shadow={opacity=.14,shadow xshift=.35mm,shadow yshift=-.45mm}},
  tarjeta/.style={rounded corners=1.4mm,fill=white,draw=zline,line width=.5pt,sombra,inner sep=2mm,align=center},
  ficha/.style={rounded corners=.9mm,line width=.4pt,draw=zmute!55,fill=#1,sombra},
  ficha/.default=zlime,
  flecha/.style={-{Stealth[length=2.2mm,width=1.8mm]},line width=.7pt,zmute},
  rot/.style={font=\sffamily\scriptsize\bfseries,text=zolive}}
\newcommand{\zt}[2][zlime!80]{\filldraw[ficha=#1] #2;}

% ---------- recta numérica (macro \recta de los capítulos) ----------
\renewcommand{\recta}[3]{\begin{tikzpicture}[x=7mm,font=\sffamily\scriptsize]
  \draw[zline,line width=2.2pt] (#1-.7,0)--(#2+.7,0);
  \draw[flecha] (#2+.5,0)--(#2+.95,0); \draw[flecha] (#1-.5,0)--(#1-.95,0);
  \foreach \t in {#1,...,#2}{\fill[white,draw=zmute,line width=.5pt] (\t,0) circle (1.1pt); \node[below=1.5mm,text=zink] at (\t,0) {$\t$};}
  \fill[zolive] (0,0) circle (1.6pt);
  #3
\end{tikzpicture}}
